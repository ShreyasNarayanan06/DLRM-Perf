cmake_minimum_required(VERSION 3.12)
project(architecture_perf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(perf_baseline_directory ../src/region.cpp)
target_include_directories(perf_baseline_directory PUBLIC ../include)
target_compile_options(perf_baseline_directory PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(performance_model model.cpp profile.cpp dlrm.cpp scenarios.cpp directory.cpp)
target_link_libraries(performance_model PUBLIC perf_baseline_directory)
target_include_directories(performance_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(performance_model PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(performance_model PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_libraries(performance_model PUBLIC -fsanitize=address,undefined)
    target_compile_options(perf_baseline_directory PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

add_executable(architecture_perf main.cpp)
target_link_libraries(architecture_perf PRIVATE performance_model)
target_compile_options(architecture_perf PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

enable_testing()
add_executable(performance_tests tests.cpp)
target_link_libraries(performance_tests PRIVATE performance_model)
target_compile_options(performance_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_test(NAME policy_neutral_performance_tests COMMAND performance_tests)
add_test(NAME policy_neutral_cli COMMAND architecture_perf)
add_test(NAME policy_neutral_invalid_bandwidth COMMAND architecture_perf --bandwidth-gbps 0)
set_tests_properties(policy_neutral_invalid_bandwidth PROPERTIES WILL_FAIL TRUE)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    find_package(Threads REQUIRED)
    add_executable(cpu_coherence_probe cpu_probe.cpp)
    target_link_libraries(cpu_coherence_probe PRIVATE Threads::Threads)
    target_compile_options(cpu_coherence_probe PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

add_executable(dlrm_tests dlrm_tests.cpp)
target_link_libraries(dlrm_tests PRIVATE performance_model)
target_compile_options(dlrm_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_test(NAME dlrm_unit_and_timing COMMAND dlrm_tests)
add_test(NAME paired_workload_cli COMMAND architecture_perf --workload both)
add_test(NAME dlrm_cli COMMAND architecture_perf --workload dlrm --rounds 8)

add_executable(directory_tests directory_tests.cpp)
target_link_libraries(directory_tests PRIVATE performance_model)
target_compile_options(directory_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_test(NAME directory_capacity_queue_and_warmup COMMAND directory_tests)
