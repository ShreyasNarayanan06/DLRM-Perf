cmake_minimum_required(VERSION 3.12)
project(architecture_perf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(performance_model model.cpp workload.cpp dlrm.cpp profile.cpp directory.cpp histograms.cpp report.cpp management.cpp comparison_report.cpp)
target_include_directories(performance_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(performance_model PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(performance_model PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_libraries(performance_model PUBLIC -fsanitize=address,undefined)
endif()
add_executable(architecture_perf main.cpp)
target_link_libraries(architecture_perf PRIVATE performance_model)
target_compile_options(architecture_perf PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
enable_testing()
add_executable(performance_tests tests.cpp)
target_link_libraries(performance_tests PRIVATE performance_model)
target_compile_options(performance_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_test(NAME baseline_timing_and_distributions COMMAND performance_tests)
add_test(NAME baseline_dlrm_cli COMMAND architecture_perf --rounds 8)
add_test(NAME baseline_paired_cli COMMAND architecture_perf --workload both --rounds 8)
add_test(NAME baseline_invalid_bandwidth COMMAND architecture_perf --bandwidth-gbps 0)
set_tests_properties(baseline_invalid_bandwidth PROPERTIES WILL_FAIL TRUE)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    find_package(Threads REQUIRED)
    add_executable(cpu_coherence_probe cpu_probe.cpp)
    target_link_libraries(cpu_coherence_probe PRIVATE Threads::Threads)
endif()
add_executable(management_tests management_tests.cpp)
target_link_libraries(management_tests PRIVATE performance_model)
target_compile_options(management_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_test(NAME management_correctness_and_transitions COMMAND management_tests)
add_test(NAME management_comparison_cli COMMAND architecture_perf --rounds 8 --management compare)
add_test(NAME management_invalid_region COMMAND architecture_perf --management dynamic --region-bytes 96)
set_tests_properties(management_invalid_region PROPERTIES WILL_FAIL TRUE)
